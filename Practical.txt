    </style>

<body>

    <h2>Student Result Calculator</h2>

    <input type="number" id="count" placeholder="Enter number of subjects">

    <button onclick="createFields()">Submit</button>

    <div id="subjects"></div>

    <div id="result"></div>

    <script>

        const createFields = () => {

            let count = Number(document.getElementById("count").value);

            let output = "";

            if (count < 2) {
                alert("Enter at least 2 subjects");
                return;
            }

            for (let i = 1; i <= count; i++) {

                output += `
                    <input type="text" id="name${i}" placeholder="Subject ${i} Name">
                    <input type="number" id="mark${i}" placeholder="Marks (0-100)">
                    <br>
                `;
            }

            output += `<br><button onclick="showResult()">Calculate Result</button>`;

            document.getElementById("subjects").innerHTML = output;
        };


        const calculate = (...marks) => {

            let total = 0;

            marks.forEach(mark => {
                total += mark;
            });

            let average = total / marks.length;

            let percentage = average;

            let status = marks.every(mark => mark >= 35);

            return `
                Total Marks: ${total}<br>
                Average: ${average.toFixed(2)}<br>
                Percentage: ${percentage.toFixed(2)}%<br>
                Result: ${status ? "PASS" : "FAIL"}
            `;
        };


        const showResult = () => {

            let count = Number(document.getElementById("count").value);

            let marks = [];

            for (let i = 1; i <= count; i++) {

                let name = document.getElementById("name" + i).value;
                let mark = document.getElementById("mark" + i).value;

                if (name === "" || mark === "") {
                    alert("Please fill all details");
                    return;
                }

                mark = Number(mark);

                if (mark < 0 || mark > 100) {
                    alert("Marks must be between 0 and 100");
                    return;
                }

                marks.push(mark);
            }

            document.getElementById("result").innerHTML = calculate(...marks);
        };

    </script>

</body>
</html>

Practical 3:Employee Bonus Eligibility

```html
<!DOCTYPE html>
<html>
<head>
    <title>Employee Bonus</title>
</head>
<body>

    <h2>Employee Bonus Eligibility</h2>

    Name: <input id="name"><br><br>
    Department: <input id="dept"><br><br>
    Salary: <input type="number" id="salary"><br><br>

    <button onclick="addEmployee()">Add Employee</button>
    <button onclick="filterEmployees()">Check Bonus</button>

    <h3>All Employees</h3>
    <table border="1" id="all"></table>

    <h3>Eligible Employees</h3>
    <table border="1" id="eligible"></table>

    <script>

        class Employee {
            constructor(name, dept, salary) {
                this.name = name;
                this.department = dept;
                this.salary = salary;
            }

            isEligibleForBonus() {
                return this.salary > 50000;
            }
        }

        let employees = [];

        function addEmployee() {
            let name = document.getElementById("name").value;
            let dept = document.getElementById("dept").value;
            let salary = Number(document.getElementById("salary").value);

            employees.push(new Employee(name, dept, salary));

            displayEmployees();
        }

        function displayEmployees() {
            let data = "<tr><th>Name</th><th>Department</th><th>Salary</th></tr>";

            employees.forEach(emp => {
                data += `<tr><td>${emp.name}</td>
                <td>${emp.department}</td>
                <td>${emp.salary}</td></tr>`;
            });

            document.getElementById("all").innerHTML = data;
        }

        function filterEmployees() {
            let eligible = employees.filter(emp => emp.isEligibleForBonus());

            let data = "<tr><th>Name</th><th>Department</th><th>Salary</th></tr>";

            eligible.forEach(emp => {
                data += `<tr><td>${emp.name}</td>
                <td>${emp.department}</td>
                <td>${emp.salary}</td></tr>`;
            });

            document.getElementById("eligible").innerHTML = data;
        }

    </script>

</body>
</html>
```

const fs = require("fs");

fs.readFile("employees.txt", "utf8", (err, data) => {

    if (err) {
        console.log(err);
        return;
    }

    let report = "";

    data.trim().split("\n").forEach((emp) => {

        let d = emp.split(",");

        let name = d[0];
        let dept = d[1];
        let salary = Number(d[2]);

        let bonus = salary * 0.1;
        let net = salary + bonus;

        report += `${name} ${dept} Salary: ${salary} Bonus: ${bonus} Net: ${net}\n`;
    });

    fs.writeFile("salaryReport.txt", report, (err) => {

        if (err)
            console.log(err);
        else
            console.log("Report created successfully!");

    });

});

Practical 5: Bank Account Deposit & Withdrawal
account.js 

let balance = 1000;

function deposit(amount) {
    balance += amount;
    return balance;
}

function withdraw(amount) {
    if (amount > balance)
        return "Insufficient Balance";

    balance -= amount;
    return balance;
}

module.exports = { deposit, withdraw };

const http = require("http");
const fs = require("fs");

http.createServer((req, res) => {
    let url = new URL(req.url, "http://localhost");

    if (url.pathname == "/") {
        res.end(fs.readFileSync("index.html"));
    }

    else if (url.pathname == "/schemes") {
        res.end(JSON.stringify([
            "Home Loan - 8.5%",
            "Personal Loan - 11.5%",
            "Education Loan - 9%"
        ]));
    }

    else if (url.pathname == "/emi") {
        let p = Number(url.searchParams.get("amount"));
        let r = Number(url.searchParams.get("rate")) / 1200;
        let n = Number(url.searchParams.get("years")) * 12;

        let emi = p * r * Math.pow(1 + r, n) /
                  (Math.pow(1 + r, n) - 1);

        res.end(JSON.stringify({emi: emi.toFixed(2)}));
    }

    else if (url.pathname == "/eligibility") {
        let income = Number(url.searchParams.get("income"));
        let loan = Number(url.searchParams.get("loan"));

        res.end(JSON.stringify({
            message: loan <= income * 10 ?
            "Eligible for Loan" : "Not Eligible"
        }));
    }

    else {
        res.statusCode = 404;
        res.end("404 Page Not Found");
    }

}).listen(3000, () => console.log("Server running on port 3000"));

const fs = require("fs");

let students = JSON.parse(fs.readFileSync("students.json"));

let topper = "";
let highestTotal = 0;
let subjectHigh = [0, 0, 0];
let failed = [];

students.forEach((s) => {
    let total = s.marks.reduce((a, b) => a + b, 0);

    let percentage = total / 3;

    let grade =
        percentage >= 75 ? "A" :
        percentage >= 60 ? "B" :
        percentage >= 40 ? "C" : "F";

    console.log("\nName:", s.name);
    console.log("Total:", total);
    console.log("Percentage:", percentage + "%");
    console.log("Grade:", grade);

    if (total > highestTotal) {
        highestTotal = total;
        topper = s.name;
    }

    s.marks.forEach((m, i) => {
        if (m > subjectHigh[i])
            subjectHigh[i] = m;
    });

    if (s.marks.some((m) => m < 35))
        failed.push(s.name);
});

console.log("\nClass Topper:", topper);

console.log("Failed Students:", failed);

Practical 9: Employee REST API using Express.js
npm init -y
npm install express
server.js

const express = require("express");
const app = express();

app.use(express.json());

// Employee Data
let employees = [
    { id: 1, name: "Prem", salary: 30000 },
    { id: 2, name: "Amit", salary: 40000 }
];

// GET - Display All Employees
app.get("/employees", (req, res) => {
    res.json(employees);
});

// POST - Add Employee
app.post("/employees", (req, res) => {
    employees.push(req.body);
    res.send("Employee Added");
});

// PUT - Update Employee
app.put("/employees/:id", (req, res) => {
    let emp = employees.find((e) => e.id == req.params.id);

    if (emp) {
        emp.name = req.body.name;
        emp.salary = req.body.salary;

        res.send("Employee Updated");
    } else {
        res.send("Employee Not Found");
    }
});

// DELETE - Delete Employee
app.delete("/employees/:id", (req, res) => {
    let old = employees.length;

    employees = employees.filter(
        (e) => e.id != req.params.id
    );

    res.send(
        old == employees.length
            ? "Employee Not Found"
            : "Employee Deleted"
    );
});

// Handle Invalid Routes
app.use((req, res) => {
    res.status(404).send("404 - Route Not Found");
});

// Start Server
app.listen(3000, () => {
    console.log("Server running on port 3000");
});
